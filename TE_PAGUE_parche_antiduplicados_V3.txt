1) INSERTAR estas funciones después de referenciaCorta(ref):

// Normaliza la referencia para comparar la referencia completa.
function normalizarReferencia(ref) {
  return String(ref || '').replace(/\D/g, '');
}

function claveDia(timestamp) {
  const fecha = new Date(Number(timestamp));
  if (!Number.isFinite(fecha.getTime())) return '';
  return [
    fecha.getFullYear(),
    String(fecha.getMonth() + 1).padStart(2, '0'),
    String(fecha.getDate()).padStart(2, '0')
  ].join('-');
}

function buscarPagoDuplicado({ref, montoNum, timestamp, idOperacion}) {
  const idNuevo = String(idOperacion || '').trim();
  const refNueva = normalizarReferencia(ref);
  const diaNuevo = claveDia(timestamp);

  return historial.find(p => {
    const idGuardado = String(p.idOperacion || '').trim();

    // El identificador bancario tiene prioridad cuando existe en ambos pagos.
    if (idNuevo && idGuardado && idNuevo === idGuardado) return true;

    // Respaldo: referencia completa + monto + mismo día.
    // No se compara la hora de activación porque MacroDroid puede cambiarla.
    return Boolean(
      refNueva &&
      diaNuevo &&
      normalizarReferencia(p.ref) === refNueva &&
      Math.round(Number(p.montoNum) * 100) ===
        Math.round(Number(montoNum) * 100) &&
      claveDia(p.timestamp) === diaNuevo
    );
  });
}

function limpiarParametrosPago() {
  window.history.replaceState({}, document.title, window.location.pathname);
}

LÍMITES Y PRECAUCIONES
----------------------
- No tocar MacroDroid V12.
- No reemplazar el HTML V3 original.
- No integrar todavía en la versión estable.
- La coincidencia de respaldo usa referencia completa, monto y día. Si el banco entrega
  un identificador de operación estable, ese identificador es preferible.
- Si dos operaciones legítimas del mismo día compartieran exactamente referencia y monto,
  el respaldo podría confundirlas. Por eso hay que verificar el comportamiento con pagos
  reales de prueba antes de aceptar esta corrección.

Este parche todavía requiere integración en una copia del HTML y pruebas en Android.
